\begin{frame}{Gleichgewicht: dynamischer Weg}
    \begin{itemize}
        \item Fall: \code{WedgeLevitatorDynamicDrop} mit \code{interFALFlow}.
        \item Workflow für eine robuste Initialisierung:
        \begin{enumerate}
            \item Kurzen Testlauf aus plausibler Tropfenposition.
            \item Tropfen nahe der beobachteten Gleichgewichtslage neu initialisieren.
            \item Flüssigkeitsviskosität temporär erhöhen $\rightarrow$ schnelle Dämpfung.
            \item Nach Relaxation auf reale Viskosität zurücksetzen.
            \item Simulation mit \code{-latestTime} fortsetzen.
        \end{enumerate}
        \item Monitoring: \code{postProcessing/dropGeometry/geometry.dat} mit Mittelpunkt, horizontaler/vertikaler Halbachse und Volumen.
        \item Vorteil: Gleichgewicht wird im vollen gekoppelten VOF-Modell geprüft.
    \end{itemize}
\end{frame}

\begin{frame}{DynamicDrop: Mesh-Fortschritt}
    \begin{itemize}
        \item Neue Beobachtung: mit gezielt verfeinertem Mesh kann die benötigte Simulationszeit deutlich reduziert werden.
        \item Lokale Verfeinerung verbessert die Tropfendynamik und kann die Relaxation zur relevanten Konfiguration beschleunigen.
        \item Tradeoff: nicht-orthogonale Zellen verschlechtern Matrixqualität, Stabilität und teilweise den Solver-Aufwand.
        \item Für weitere AMR-Studien müssen daher drei Größen gemeinsam bewertet werden:
        \begin{enumerate}
            \item physikalische Genauigkeit der Tropfendynamik,
            \item Nicht-Orthogonalität und Mesh-Qualität,
            \item Laufzeit und Skalierung des gekoppelten Solvers.
        \end{enumerate}
    \end{itemize}
\end{frame}

\begin{frame}{Gleichgewicht: Pseudo-Solver}
    \begin{columns}
    \column{.55\linewidth}
    \begin{itemize}
        \item Solver: \code{acousticDropEquilibriumFoam}.
        \item Idee nach dem Stabilitäts-Workflow von Andrade et al. \citep{andrade2019numerical}:
        \begin{enumerate}
            \item Akustisches Feld für feste Tropfenform lösen.
            \item Kraft auf \code{dropWall} integrieren.
            \item Vertikale Position bis $F_y \approx \rho_l V g$ korrigieren.
            \item Formresiduum aus
            \[
                p_r+\rho_l g y+\gamma\kappa = h
            \]
            auswerten.
            \item Normalen-Update mit Volumenkorrektur, dann wiederholen.
        \end{enumerate}
    \end{itemize}
